-- 1) TABELA
CREATE TABLE public.tb_4002_especie (
    id              SERIAL PRIMARY KEY,
    codigo          VARCHAR(3) NOT NULL UNIQUE,
    descricao       VARCHAR(60) NOT NULL,
    criado_em       TIMESTAMP NOT NULL DEFAULT NOW(),
    atualizado_em   TIMESTAMP NOT NULL DEFAULT NOW()
);

COMMENT ON TABLE public.tb_4002_especie IS 'Catalogo de Especie - origem GR4002 (GERAS)';

-- 2) TRIGGER DEDICADA (atualizado_em)
CREATE OR REPLACE FUNCTION public.trg_tb_4002_especie_atualizado_em()
RETURNS TRIGGER AS $$
BEGIN
    NEW.atualizado_em = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_tb_4002_especie_atualizado_em
BEFORE UPDATE ON public.tb_4002_especie
FOR EACH ROW
EXECUTE FUNCTION public.trg_tb_4002_especie_atualizado_em();

-- 3) SEED (dados reais do GR4002)

--     SIMULADO       - conector de mentira (nao sai do Orion), para testes.

CREATE TABLE IF NOT EXISTS public.tb_blk_8000_provedores (
    id          SERIAL PRIMARY KEY,
    codigo      VARCHAR(30)  NOT NULL UNIQUE,
    nome        VARCHAR(80)  NOT NULL,
    ativo       BOOLEAN      NOT NULL DEFAULT TRUE,
    criado_em   TIMESTAMP    NOT NULL DEFAULT now()
);

INSERT INTO public.tb_blk_8000_provedores (codigo, nome, ativo) VALUES
    ('SIMULADO',      'Simulado (teste interno, sem rede)', TRUE),
    ('STELLAR_TESTE', 'Stellar - rede de teste',            TRUE),
    ('POLYGON_TESTE', 'Polygon - rede de teste',            TRUE),
    ('DEFARM',        'DeFarm (Stellar)',                   FALSE)
ON CONFLICT (codigo) DO NOTHING;

-- Conferencia
SELECT id, codigo, nome, ativo FROM public.tb_blk_8000_provedores ORDER BY id;

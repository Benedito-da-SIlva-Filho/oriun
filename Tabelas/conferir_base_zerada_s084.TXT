--   protocolos 4, tb_1007_motivo_mortes 44.

--   avise.

SELECT * FROM (
  SELECT table_schema, table_name,
         (xpath('/row/c/text()',
                query_to_xml(format('select count(*) as c from %I.%I', table_schema, table_name), false, true, '')))[1]::text::int AS linhas
  FROM information_schema.tables
  WHERE table_type = 'BASE TABLE'
    AND table_schema IN ('produtor_00001', 'produtor_00004', 'produtor_00005', 'produtor_00010', 'public')
) t
WHERE linhas > 0
ORDER BY table_schema, table_name;

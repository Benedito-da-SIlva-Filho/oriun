-- 1) TABELA
CREATE TABLE public.tb_4005_racas (
    id              SERIAL PRIMARY KEY,
    codigo          VARCHAR(2) NOT NULL UNIQUE,
    descricao       VARCHAR(60) NOT NULL,
    criado_em       TIMESTAMP NOT NULL DEFAULT NOW(),
    atualizado_em   TIMESTAMP NOT NULL DEFAULT NOW()
);

COMMENT ON TABLE public.tb_4005_racas IS 'Catalogo oficial de Racas SISBOV - origem GR4005 (GERAS). Registro AA do legado intencionalmente fora do seed.';

-- 2) TRIGGER DEDICADA (atualizado_em)
CREATE OR REPLACE FUNCTION public.trg_tb_4005_racas_atualizado_em()
RETURNS TRIGGER AS $$
BEGIN
    NEW.atualizado_em = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_tb_4005_racas_atualizado_em
BEFORE UPDATE ON public.tb_4005_racas
FOR EACH ROW
EXECUTE FUNCTION public.trg_tb_4005_racas_atualizado_em();

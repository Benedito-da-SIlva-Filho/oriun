CREATE TABLE tb_5001_brincagem (
    id              SERIAL PRIMARY KEY,

    id_gta          INTEGER NOT NULL,
    id_propriedade  INTEGER NOT NULL,
    id_produtor     INTEGER NOT NULL,

    sexo            VARCHAR(10) NOT NULL,      -- MACHO / FEMEA
    raca            VARCHAR(50) NULL,
    idade_meses     VARCHAR(20) NOT NULL,       -- faixa texto, ex: "13 A 24 MESES"

    sisbov          VARCHAR(15) NULL,           -- em branco ate o Peao ler o brinco
    rfid            VARCHAR(30) NULL,
    manejo          VARCHAR(6) NULL,            -- calculado automaticamente (regra 105/076)

    lote            VARCHAR(100) NULL,          -- piquete/pasto, texto livre
    saida_brincos   VARCHAR(3) NULL,            -- 'COM' / 'SEM'

    sts             BOOLEAN NOT NULL DEFAULT FALSE,  -- False = pendente de Comunicado
    id_comunicado   INTEGER NULL,               -- carimbado no "Gerar Comunicado"

    criado_em       TIMESTAMP NOT NULL DEFAULT NOW(),
    atualizado_em   TIMESTAMP NOT NULL DEFAULT NOW(),

    CONSTRAINT fk_tb_5001_brincagem_gta
        FOREIGN KEY (id_gta) REFERENCES tb_4001_gta(id),
    CONSTRAINT fk_tb_5001_brincagem_propriedade
        FOREIGN KEY (id_propriedade) REFERENCES propriedades(id),
    CONSTRAINT fk_tb_5001_brincagem_produtor
        FOREIGN KEY (id_produtor) REFERENCES produtores(id),

    CONSTRAINT chk_tb_5001_brincagem_saida_brincos
        CHECK (saida_brincos IS NULL OR saida_brincos IN ('COM', 'SEM'))
);

-- Sisbov so pode se repetir se for NULL (varias linhas vazias da mesma
-- categoria) - uma vez preenchido, deve ser unico no sistema.
CREATE UNIQUE INDEX idx_tb_5001_brincagem_sisbov ON tb_5001_brincagem (sisbov)
    WHERE sisbov IS NOT NULL;

CREATE INDEX idx_tb_5001_brincagem_id_gta ON tb_5001_brincagem (id_gta);
CREATE INDEX idx_tb_5001_brincagem_id_propriedade ON tb_5001_brincagem (id_propriedade);
CREATE INDEX idx_tb_5001_brincagem_id_produtor ON tb_5001_brincagem (id_produtor);
CREATE INDEX idx_tb_5001_brincagem_sts ON tb_5001_brincagem (sts);
-- Apoio direto para o "Quadro" do Brete: contagem de linhas vazias por categoria
CREATE INDEX idx_tb_5001_brincagem_categoria_pendente
    ON tb_5001_brincagem (id_gta, sexo, idade_meses)
    WHERE sisbov IS NULL;

CREATE OR REPLACE FUNCTION trg_tb_5001_brincagem_atualizado_em()
RETURNS TRIGGER AS $$
BEGIN
    NEW.atualizado_em = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_tb_5001_brincagem_atualizado_em
BEFORE UPDATE ON tb_5001_brincagem
FOR EACH ROW
EXECUTE FUNCTION trg_tb_5001_brincagem_atualizado_em();

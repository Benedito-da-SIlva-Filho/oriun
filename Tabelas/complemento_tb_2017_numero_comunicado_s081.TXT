-- =========================================================
-- PASSO 1 - Previa: quantas linhas seriam completadas no Zopone
-- (troque o schema para conferir outro Produtor)
-- =========================================================
SELECT COUNT(*) AS linhas_a_completar
FROM produtor_00004.tb_2017_movimentacoes m
JOIN produtor_00004.tb_2001 e     ON e.id = m.id_animal
JOIN produtor_00004.tb_4001_gta g ON g.id = e.id_gta_entrada
WHERE e.numero_comunicado IS NOT NULL
  AND m.numero_comunicado IS NULL
  AND m.tipo_saida IS NULL
  AND m.gta_entrada = g.numero_gta_completo;


-- =========================================================
-- PASSO 2 - Complemento em todos os schemas de Produtor
-- (o resultado aparece na aba Messages)
-- =========================================================
DO $$
DECLARE
    r       RECORD;
    v_qtde  INTEGER;
BEGIN
    FOR r IN
        SELECT t.table_schema AS esquema
        FROM information_schema.tables t
        WHERE t.table_schema LIKE 'produtor\_%'
          AND t.table_name = 'tb_2017_movimentacoes'
        ORDER BY t.table_schema
    LOOP
        EXECUTE format($sql$
            UPDATE %1$I.tb_2017_movimentacoes m
               SET numero_comunicado = lpad(e.numero_comunicado::text, 6, '0')
              FROM %1$I.tb_2001 e
              JOIN %1$I.tb_4001_gta g ON g.id = e.id_gta_entrada
             WHERE e.id = m.id_animal
               AND e.numero_comunicado IS NOT NULL
               AND m.numero_comunicado IS NULL
               AND m.tipo_saida IS NULL
               AND m.gta_entrada = g.numero_gta_completo
        $sql$, r.esquema);
        GET DIAGNOSTICS v_qtde = ROW_COUNT;

        RAISE NOTICE '% : % linha(s) do kardex completada(s)', r.esquema, v_qtde;
    END LOOP;
END $$;

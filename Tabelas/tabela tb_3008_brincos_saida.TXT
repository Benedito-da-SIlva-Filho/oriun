--        COMPRA, NASCTO-DESMAME, REIDENTIFICACAO, TRANSFERENCIA.

-- ENUM compartilhado para Tipo Entrada / Origem (usado tambem em outras tabelas do modulo Animais,
-- como tb_2001, tb_2017, tb_5001, quando forem criadas)
DO $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'tipo_entrada_enum') THEN
        CREATE TYPE tipo_entrada_enum AS ENUM (
            'COMPRA',
            'NASCTO-DESMAME',
            'REIDENTIFICACAO',
            'TRANSFERENCIA'
        );
    END IF;
END $$;

CREATE TABLE tb_3008_brincos_saida (
    id                  SERIAL PRIMARY KEY,
    id_produtor         INTEGER NOT NULL REFERENCES produtores(id),
    id_propriedade      INTEGER NOT NULL REFERENCES propriedades(id),
    id_lote_entrada     INTEGER NOT NULL REFERENCES tb_3007_brincos_entrada(id),
    sisbov              VARCHAR(15) NOT NULL,
    rfid                VARCHAR(15),
    data_saida          DATE,
    gta                 VARCHAR(10),
    origem              tipo_entrada_enum,
    criado_em           TIMESTAMP NOT NULL DEFAULT NOW(),
    atualizado_em       TIMESTAMP NOT NULL DEFAULT NOW()
);

-- Indices de apoio
CREATE INDEX idx_tb_3008_brincos_saida_produtor
    ON tb_3008_brincos_saida (id_produtor);

CREATE INDEX idx_tb_3008_brincos_saida_propriedade
    ON tb_3008_brincos_saida (id_propriedade);

CREATE INDEX idx_tb_3008_brincos_saida_sisbov
    ON tb_3008_brincos_saida (sisbov);

CREATE INDEX idx_tb_3008_brincos_saida_lote_entrada
    ON tb_3008_brincos_saida (id_lote_entrada);

-- Trigger dedicado para atualizado_em (padrao: trg_<tablename>_atualizado_em)
CREATE OR REPLACE FUNCTION trg_tb_3008_brincos_saida_atualizado_em()
RETURNS TRIGGER AS $$
BEGIN
    NEW.atualizado_em = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_tb_3008_brincos_saida_atualizado_em
    BEFORE UPDATE ON tb_3008_brincos_saida
    FOR EACH ROW
    EXECUTE FUNCTION trg_tb_3008_brincos_saida_atualizado_em();

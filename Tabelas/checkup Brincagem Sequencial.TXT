-- 1) tb_2001 — deve dar total=50, sem_especie=0, sem_propriedade=0
SELECT
    count(*) AS total,
    count(*) FILTER (WHERE id_especie IS NULL) AS sem_especie,
    count(*) FILTER (WHERE id_propriedade IS NULL) AS sem_propriedade,
    count(*) FILTER (WHERE id_raca IS NULL) AS sem_raca
FROM tb_2001
WHERE id_gta_entrada = 10;

-- 3) tb_7017 — os 50 brincos usados devem aparecer com status atualizado e data_brincagem preenchida

FROM tb_7017_entrada_brincos_rfid

CREATE TABLE tb_1021_fabricantes_brincos (
    id              SERIAL PRIMARY KEY,
    nome            VARCHAR(50)  NOT NULL,
    logo            VARCHAR(100),
    endereco        VARCHAR(50),
    complemento     VARCHAR(50),
    bairro          VARCHAR(40),
    cidade          VARCHAR(40),
    uf              VARCHAR(2),
    cep             VARCHAR(8),
    ativo           BOOLEAN NOT NULL DEFAULT TRUE,
    criado_em       TIMESTAMP NOT NULL DEFAULT NOW(),
    atualizado_em   TIMESTAMP NOT NULL DEFAULT NOW()
);

-- Indice de apoio para busca por nome (tela de cadastro/consulta)
CREATE INDEX idx_tb_1021_fabricantes_brincos_nome
    ON tb_1021_fabricantes_brincos (nome);

-- Trigger dedicado para atualizado_em (padrao: trg_<tablename>_atualizado_em)
CREATE OR REPLACE FUNCTION trg_tb_1021_fabricantes_brincos_atualizado_em()
RETURNS TRIGGER AS $$
BEGIN
    NEW.atualizado_em = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_tb_1021_fabricantes_brincos_atualizado_em
    BEFORE UPDATE ON tb_1021_fabricantes_brincos
    FOR EACH ROW
    EXECUTE FUNCTION trg_tb_1021_fabricantes_brincos_atualizado_em();

--     IDEMPOTENTE: ADD COLUMN IF NOT EXISTS.

DO $$
DECLARE
    schema_rec RECORD;
BEGIN
    FOR schema_rec IN
        SELECT schema_name FROM information_schema.schemata
        WHERE schema_name LIKE 'produtor_%'
    LOOP
        EXECUTE format(
            'ALTER TABLE %I.tb_2001 ADD COLUMN IF NOT EXISTS data_brincagem DATE',
            schema_rec.schema_name
        );
    END LOOP;
END $$;

CREATE TABLE tb_3007_brincos_entrada (
    id                  SERIAL PRIMARY KEY,
    id_produtor         INTEGER NOT NULL REFERENCES produtores(id),
    id_propriedade      INTEGER NOT NULL REFERENCES propriedades(id),
    id_fabricante       INTEGER NOT NULL REFERENCES tb_1021_fabricantes_brincos(id),
    id_responsavel      INTEGER REFERENCES usuarios(id),
    sequencia           INTEGER,
    codigo_inicial      VARCHAR(15) NOT NULL,
    codigo_final        VARCHAR(15) NOT NULL,
    nota_fiscal         VARCHAR(6),
    qtde_entrada        INTEGER NOT NULL,
    data_entrada        DATE,
    qtde_saida          INTEGER NOT NULL DEFAULT 0,
    saldo               INTEGER NOT NULL DEFAULT 0,
    solicitacao_nro     INTEGER,
    solicitacao_data    DATE,
    data_fabricacao     DATE,
    total_caixas        INTEGER,
    criado_em           TIMESTAMP NOT NULL DEFAULT NOW(),
    atualizado_em       TIMESTAMP NOT NULL DEFAULT NOW()
);

-- Indices de apoio para as buscas mais comuns (por Produtor/Propriedade e por faixa de codigo)

-- Trigger dedicado para atualizado_em (padrao: trg_<tablename>_atualizado_em)

    NEW.atualizado_em = NOW();
    RETURN NEW;

CREATE TRIGGER trg_tb_3007_brincos_entrada_atualizado_em

-- 1) Tabelas que o produtor_00004 realmente tem
SELECT table_name FROM information_schema.tables
WHERE table_schema = 'produtor_00004' ORDER BY 1;

-- 2) Tabelas do schema público
SELECT table_name FROM information_schema.tables
WHERE table_schema = 'public' AND table_name LIKE 'tb\_%' ORDER BY 1;

-- 3) Todos os produtores têm o mesmo número de tabelas?
SELECT table_schema, count(*) FROM information_schema.tables
WHERE table_schema LIKE 'produtor\_%' GROUP BY 1 ORDER BY 1;

SELECT column_name, data_type, column_default, is_nullable
FROM information_schema.columns
WHERE table_schema = 'produtor_00004' AND table_name = 'tb_versao_schema'
ORDER BY ordinal_position;

SELECT column_name, data_type, column_default, is_nullable
FROM information_schema.columns
WHERE table_name = 'tb_versao_schema'
ORDER BY table_schema, ordinal_position;
